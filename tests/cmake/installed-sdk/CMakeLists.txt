cmake_minimum_required(VERSION 3.30)
project(ascii-chat-sdk-consumer LANGUAGES C)
set(SDK_COMPONENT shared CACHE STRING "Installed SDK component to exercise")
set(SDK_PACKAGE asciichat CACHE STRING "Public package spelling")
find_package(${SDK_PACKAGE} 0.3 CONFIG REQUIRED COMPONENTS ${SDK_COMPONENT})
add_executable(sdk-consumer main.c)
target_link_libraries(sdk-consumer PRIVATE asciichat::${SDK_COMPONENT})
if(WIN32 AND DEFINED ${SDK_PACKAGE}_MSVC_RUNTIME_LIBRARY)
    set_property(TARGET sdk-consumer PROPERTY MSVC_RUNTIME_LIBRARY "${${SDK_PACKAGE}_MSVC_RUNTIME_LIBRARY}")
endif()
enable_testing()
add_test(NAME installed-sdk COMMAND sdk-consumer)
