cmake_minimum_required(VERSION 3.25)
project(ascii_chat_simd_tests C)
enable_testing()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_C_STANDARD 17)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
    set(ASCIICHAT_IS_X86_64 TRUE)
    add_compile_options(-march=x86-64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
    set(ASCIICHAT_IS_ARM64 TRUE)
    add_compile_options(-march=armv8-a)
endif()
include("${REPO_ROOT}/cmake/compiler/SIMD.cmake")
add_executable(simd-test test_dispatch.c
    "${REPO_ROOT}/lib/platform/cpu.c"
    "${REPO_ROOT}/lib/video/ascii/simd/dispatch.c")
target_include_directories(simd-test PRIVATE "${REPO_ROOT}/include" "${REPO_ROOT}/deps")
if(ENABLE_SIMD_AVX2)
    target_sources(simd-test PRIVATE "${REPO_ROOT}/lib/video/ascii/avx2/common.c")
    set_source_files_properties("${REPO_ROOT}/lib/video/ascii/avx2/common.c" PROPERTIES COMPILE_OPTIONS "-mavx2;-fno-lto")
endif()
foreach(isa SSE2 SSSE3 AVX2 NEON SVE)
    if(ENABLE_SIMD_${isa})
        string(TOLOWER "${isa}" name)
        set(source "${REPO_ROOT}/lib/video/ascii/${name}/luminance.c")
        if(isa STREQUAL "SVE")
            set(flags -march=armv8-a+sve)
        elseif(isa STREQUAL "NEON")
            set(flags -march=armv8-a)
        else()
            set(flags -m${name})
        endif()
        set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS "${flags};-fno-lto")
        target_sources(simd-test PRIVATE "${source}")
    endif()
endforeach()
add_test(NAME simd-dispatch COMMAND simd-test)
add_test(NAME simd-forced-scalar COMMAND simd-test scalar)
set_tests_properties(simd-forced-scalar PROPERTIES ENVIRONMENT "ASCII_CHAT_SIMD=scalar")
add_test(NAME simd-invalid-override COMMAND simd-test scalar)
set_tests_properties(simd-invalid-override PROPERTIES ENVIRONMENT "ASCII_CHAT_SIMD=invalid")

option(TEST_EMISSION "Build isolated production renderer tests (Linux)" OFF)
if(TEST_EMISSION)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "TEST_EMISSION uses the Linux standalone runtime; use Criterion on other Unix platforms")
    endif()
    add_executable(emission-test ../unit/video/ascii_emission_test.c emission_runtime.c)
    target_compile_definitions(emission-test PRIVATE NDEBUG _GNU_SOURCE ASCII_EMISSION_STANDALONE)
    target_compile_options(emission-test PRIVATE -ffunction-sections -fdata-sections)
    target_link_options(emission-test PRIVATE -Wl,--gc-sections)
    target_include_directories(emission-test PRIVATE "${REPO_ROOT}/include" "${REPO_ROOT}/deps")
    target_link_libraries(emission-test PRIVATE m pthread)
    set(emission_sources
        lib/video/ascii/scalar/foreground.c lib/video/ascii/scalar/background.c
        lib/video/ascii/scalar/halfblock.c lib/video/ascii/output_buffer.c
        lib/video/ascii/simd/cells.c lib/video/ascii/render.c lib/video/ascii/sgr.c lib/video/ascii/common.c lib/video/ascii/palette.c
        lib/video/ascii/ascii.c
        lib/video/terminal/ansi.c
        lib/video/ascii/simd/dispatch.c lib/platform/cpu.c
        lib/lifecycle.c lib/atomic.c lib/platform/rwlock.c
        lib/platform/posix/rwlock.c lib/platform/posix/mutex.c
        lib/platform/posix/string.c lib/platform/posix/system.c lib/platform/system.c lib/util/time.c)
    foreach(source IN LISTS emission_sources)
        target_sources(emission-test PRIVATE "${REPO_ROOT}/${source}")
    endforeach()
    foreach(isa SSE2 SSSE3 AVX2 NEON SVE)
        if(ENABLE_SIMD_${isa})
            string(TOLOWER "${isa}" name)
            target_sources(emission-test PRIVATE
                "${REPO_ROOT}/lib/video/ascii/${name}/luminance.c"
                "${REPO_ROOT}/lib/video/ascii/${name}/mono.c"
                "${REPO_ROOT}/lib/video/ascii/${name}/color.c"
                "${REPO_ROOT}/lib/video/ascii/${name}/halfblock.c")
        endif()
    endforeach()
    if(ENABLE_SIMD_AVX2)
        target_sources(emission-test PRIVATE "${REPO_ROOT}/lib/video/ascii/avx2/common.c")
        set_source_files_properties("${REPO_ROOT}/lib/video/ascii/avx2/common.c" PROPERTIES COMPILE_OPTIONS "-mavx2;-fno-lto")
    endif()
    foreach(isa SSE2 SSSE3 AVX2 NEON SVE)
        if(ENABLE_SIMD_${isa})
            string(TOLOWER "${isa}" name)
            if(isa STREQUAL "SVE")
                set(flags -march=armv8-a+sve)
            elseif(isa STREQUAL "NEON")
                set(flags -march=armv8-a)
            else()
                set(flags -m${name})
            endif()
            foreach(kind mono color halfblock)
                set_source_files_properties("${REPO_ROOT}/lib/video/ascii/${name}/${kind}.c"
                    PROPERTIES COMPILE_OPTIONS "${flags};-fno-lto")
            endforeach()
        endif()
    endforeach()
    add_test(NAME simd-emission COMMAND emission-test)
    add_executable(emission-benchmark benchmark_emission.c emission_runtime.c)
    get_target_property(emission_target_sources emission-test SOURCES)
    list(FILTER emission_target_sources EXCLUDE REGEX "ascii_emission_test.c|emission_runtime.c")
    add_executable(halfblock-benchmark benchmark_halfblock.c emission_runtime.c ${emission_target_sources})
    target_include_directories(halfblock-benchmark PRIVATE "${REPO_ROOT}/include" "${REPO_ROOT}/deps")
    target_compile_definitions(halfblock-benchmark PRIVATE NDEBUG _GNU_SOURCE)
    target_compile_options(halfblock-benchmark PRIVATE -ffunction-sections -fdata-sections)
    target_link_options(halfblock-benchmark PRIVATE -Wl,--gc-sections
        -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
    target_link_libraries(halfblock-benchmark PRIVATE m pthread)
    set(EMISSION_BENCHMARK_BASELINE "" CACHE PATH "Optional original renderer source tree for before/after measurement")
    if(EMISSION_BENCHMARK_BASELINE)
        target_include_directories(emission-benchmark BEFORE PRIVATE "${EMISSION_BENCHMARK_BASELINE}/include")
        set(baseline_sources)
        foreach(source IN LISTS emission_target_sources)
            string(REPLACE "${REPO_ROOT}/" "" relative "${source}")
            if(EXISTS "${EMISSION_BENCHMARK_BASELINE}/${relative}")
                set(source "${EMISSION_BENCHMARK_BASELINE}/${relative}")
            endif()
            list(APPEND baseline_sources "${source}")
        endforeach()
        set(emission_target_sources ${baseline_sources})
        foreach(isa SSE2 SSSE3 AVX2 NEON SVE)
            if(ENABLE_SIMD_${isa})
                string(TOLOWER "${isa}" name)
                if(isa STREQUAL "SVE")
                    set(flags -march=armv8-a+sve)
                elseif(isa STREQUAL "NEON")
                    set(flags -march=armv8-a)
                else()
                    set(flags -m${name})
                endif()
                foreach(kind luminance mono color common)
                    set(source "${EMISSION_BENCHMARK_BASELINE}/lib/video/ascii/${name}/${kind}.c")
                    if(EXISTS "${source}")
                        set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS "${flags};-fno-lto")
                        if(kind STREQUAL "common")
                            list(APPEND emission_target_sources "${source}")
                        endif()
                    endif()
                endforeach()
            endif()
        endforeach()
    endif()
    target_sources(emission-benchmark PRIVATE ${emission_target_sources})
    target_include_directories(emission-benchmark PRIVATE "${REPO_ROOT}/include" "${REPO_ROOT}/deps")
    target_compile_definitions(emission-benchmark PRIVATE NDEBUG _GNU_SOURCE)
    target_compile_options(emission-benchmark PRIVATE -ffunction-sections -fdata-sections)
    target_link_options(emission-benchmark PRIVATE -Wl,--gc-sections
        -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
    target_link_libraries(emission-benchmark PRIVATE m pthread)
endif()
