cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ascii_defer_host LANGUAGES CXX)
    set(_ascii_defer_standalone TRUE)
elseif(NOT CMAKE_CXX_COMPILER_LOADED)
    enable_language(CXX)
endif()

if(NOT ASCIICHAT_LLVM_DEVELOPMENT_MODULE)
    get_filename_component(ASCIICHAT_LLVM_DEVELOPMENT_MODULE
        "${CMAKE_CURRENT_LIST_DIR}/../../../cmake/deps/LLVMDevelopment.cmake" ABSOLUTE)
endif()
include("${ASCIICHAT_LLVM_DEVELOPMENT_MODULE}")
ascii_find_llvm_development()

add_library(ascii-defer-core STATIC driver.cpp plan.cpp transform.cpp)
set_target_properties(ascii-defer-core PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
ascii_configure_llvm_host_tool(ascii-defer-core)
target_link_libraries(ascii-defer-core PUBLIC ascii-llvm-support ascii-libclang ${CMAKE_DL_LIBS})

add_executable(ascii-defer-native main.cpp)
set_target_properties(ascii-defer-native PROPERTIES OUTPUT_NAME defer-tool CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(APPLE)
    set_property(TARGET ascii-defer-native APPEND PROPERTY BUILD_RPATH "$<TARGET_FILE_DIR:ascii-libclang>")
endif()
ascii_configure_llvm_host_tool(ascii-defer-native)
target_link_libraries(ascii-defer-native PRIVATE ascii-defer-core)
if(WIN32)
    add_custom_command(TARGET ascii-defer-native POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:ascii-libclang>" "$<TARGET_FILE_DIR:ascii-defer-native>"
        VERBATIM)
endif()

if(_ascii_defer_standalone)
    include(CTest)
endif()
if((_ascii_defer_standalone AND BUILD_TESTING) OR (NOT _ascii_defer_standalone AND BUILD_TESTS))
    foreach(_test text clang transform)
        add_executable("ascii-defer-${_test}-test" "${_test}_test.cpp")
        set_target_properties("ascii-defer-${_test}-test" PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
        if(APPLE)
            set_property(TARGET "ascii-defer-${_test}-test" APPEND PROPERTY BUILD_RPATH "$<TARGET_FILE_DIR:ascii-libclang>")
        endif()
        ascii_configure_llvm_host_tool("ascii-defer-${_test}-test")
        target_link_libraries("ascii-defer-${_test}-test" PRIVATE ascii-defer-core)
        if(WIN32)
            add_custom_command(TARGET "ascii-defer-${_test}-test" POST_BUILD
                COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:ascii-libclang>" "$<TARGET_FILE_DIR:ascii-defer-${_test}-test>"
                VERBATIM)
        endif()
        add_test(NAME "ascii-defer-${_test}" COMMAND "ascii-defer-${_test}-test")
    endforeach()
endif()
